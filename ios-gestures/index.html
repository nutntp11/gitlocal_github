Tap

แตะครั้งเดียว

9:41

สั่ง AI ว่าTap an app icon to open it, with a press-down feedback.

gestures/tap.js
Gestures.tap(iconEl, {
  onTap(e, { x, y }) { openApp(); },
  maxMove: 10,       // ลากเกินนี้ = ไม่นับ
  maxDuration: 500,  // กดนานเกินนี้ = ไม่นับ
});
/* CSS: .is-pressed { transform: scale(.9) } */

9:41

กล่องจดหมาย

เก็บถาวรแล้ว

สั่ง AI ว่าSwipe a row left to reveal Archive and Delete; a full swipe deletes it.

gestures/swipe-actions.js
Gestures.swipeActions(rowEl, {
  actions: [
    { label: 'เก็บ', color: '#af52de', onAction(row, close) { close(); } },
    { label: 'ลบ',  color: '#ff3b30', onAction(row) { row.remove(); } },
  ],
  fullSwipe: true,   // ปัดยาว = ทำปุ่มสุดท้าย
});

  backdrop: backdropEl,
  scroller: sheetEl.querySelector('.sheet-body'),
  onDismiss() { showReopenButton(); },
});
sheet.open();
